# Repro: dyntopo faces not drawing (holes that reappear when re-sculpted).
make_cube subdivs=24 size=0.5
triangulate
build_spatial leaf_limit=256 depth_limit=12
set_brush_tool tool=draw
set_brush radius=0.30 strength=0.5 spacing=0.25
dyntopo enabled=1 detail=0.03 mode=both flip=1
# Sweep a stroke across the front face so dyntopo refines a band.
stroke_path p1=-0.45,0,0.5 p2=0.45,0,0.5 normal=0,0,1 steps=16
screenshot view=front out=dyntopo_missing_after_stroke.png
dump_state out=dyntopo_missing_state.json mesh=1 spatial=1
# Re-stroke the same band; the bug-report says missing faces reappear here.
stroke_path p1=-0.45,0,0.5 p2=0.45,0,0.5 normal=0,0,1 steps=16
screenshot view=front out=dyntopo_missing_after_restroke.png
dump_state out=dyntopo_missing_state2.json mesh=1 spatial=1
