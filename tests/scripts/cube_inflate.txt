# Cube with a single inflate-brush stroke at the +Z face center.
# Each vertex pushes along its own normal proportional to falloff*mask.
make_cube subdivs=12 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=inflate
set_brush radius=0.25 strength=0.5
stroke origin=0,0,0.25 normal=0,0,1
dump_state out=cube_inflate.json
screenshot view=persp out=cube_inflate.png
