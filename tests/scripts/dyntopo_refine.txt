# Dynamic topology: refine a triangulated cube under one brush dab.
# The +Z face (z=0.25 for size=0.5) is locally subdivided to the detail size;
# the mesh must remain manifold. (plan: documentation/plans/dynamic-topology.md)
make_cube subdivs=6 size=0.5
triangulate
build_spatial leaf_limit=256 depth_limit=8
set_brush radius=0.2 strength=0.0
dump_state out=dyntopo_before.json mesh=1
dyntopo enabled=1 detail=0.04 mode=subdivide
stroke origin=0,0,0.25 normal=0,0,1
assert_manifold
dump_state out=dyntopo_after.json mesh=1
screenshot view=persp out=dyntopo_refine.png
