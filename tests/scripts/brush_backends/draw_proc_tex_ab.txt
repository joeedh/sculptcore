# A/B script: set_texture proc=<name>, cpp vs wgsl backend gate.
# Bakes the `radial` analytic generator (1 - 2*dist-from-center, clamped) onto
# the UV grid and samples it in GLOBAL space. Like the pattern modes the result
# is just texels in brush.tex_pixels, so this gates the proc-generated buffer
# the same way draw_global_tex_ab gates pattern=rampx. See draw_global_tex_ab.txt.
make_cube subdivs=16 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=draw
set_brush radius=0.15 strength=6.0
set_texture proc=radial width=64 height=64
set_coord_space space=global

set_backend backend=cpp
stroke_path p1=-0.2,0,0.25 p2=0.2,0,0.25 steps=8
dump_state out=draw_proc_tex_cpp.json

undo

set_backend backend=wgsl
stroke_path p1=-0.2,0,0.25 p2=0.2,0,0.25 steps=8
dump_state out=draw_proc_tex_wgsl.json
