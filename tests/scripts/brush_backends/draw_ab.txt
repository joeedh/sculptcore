# A/B script: same draw stroke under the cpp then wgsl backend gate.
# Both currently execute through the C++ executor (no native WebGPU
# dispatch yet), so the two dumps must match exactly; the cpp dump is
# also diffed against tests/golden/draw.json by `make.mjs sbrush-verify`.
make_cube subdivs=12 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=draw
set_brush radius=0.25 strength=0.5

set_backend backend=cpp
stroke origin=0,0,0.25 normal=0,0,1
dump_state out=draw_cpp.json

undo

set_backend backend=wgsl
stroke origin=0,0,0.25 normal=0,0,1
dump_state out=draw_wgsl.json
