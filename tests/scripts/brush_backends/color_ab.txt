# A/B script: color paint, cpp vs wgsl backend gate. The color brush writes a
# per-vertex float4 "color" attribute (no vertex motion), so the meaningful
# signal is the attr fingerprint state_dump now emits — it must match across
# backends, which exercises the DSL attribute read/write path on the GPU.
#
# Full reset between backends (re-make_cube) rather than `undo`: undo doesn't
# restore the painted attr (meshlog snapshots co/no, not custom layers), but a
# fresh mesh + default-zeroed color layer gives both strokes the same
# deterministic start, so cpp == wgsl and the golden is reproducible.
make_cube subdivs=12 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=color
set_brush radius=0.25 strength=1.0
set_backend backend=cpp
stroke origin=0,0,0.25 normal=0,0,1
dump_state out=color_cpp.json

make_cube subdivs=12 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=color
set_brush radius=0.25 strength=1.0
set_backend backend=wgsl
stroke origin=0,0,0.25 normal=0,0,1
dump_state out=color_wgsl.json
