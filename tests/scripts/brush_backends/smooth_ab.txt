# A/B script: smooth stroke, cpp vs wgsl backend gate. See draw_ab.txt.
# An inflate stroke runs first (committed) to create a spike for the
# smooth stroke to relax; the A/B subject is the smooth stroke, and the
# undo reverts only it so the wgsl pass starts from the same inflated
# state.
make_cube subdivs=16 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=inflate
set_brush radius=0.05 strength=4.0
stroke origin=0,0,0.25 normal=0,0,1

set_brush_tool tool=smooth
set_brush radius=0.15 strength=1.0

set_backend backend=cpp
stroke origin=0,0,0.25 normal=0,0,1
dump_state out=smooth_cpp.json

undo

set_backend backend=wgsl
stroke origin=0,0,0.25 normal=0,0,1
dump_state out=smooth_wgsl.json
