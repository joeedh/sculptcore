# A/B script: VIEWREPEAT texture coord space, cpp vs wgsl backend gate.
# Same render matrix as draw_viewplane_tex_ab, but uv is additionally scaled by
# tex_repeat (=2.0), pushing coords outside [0,1] so the clamp-to-edge path in
# both samplers is exercised. Parity proves the render-matrix marshal AND the
# tex_repeat multiply agree across backends. See draw_ab.txt for the harness.
make_cube subdivs=16 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=draw
set_brush radius=0.15 strength=6.0
set_texture pattern=rampx width=64 height=64
set_coord_space space=viewrepeat repeat=2.0
set_render_matrix m=1.29903811,-0.75,0,0.5,0.75,1.29903811,0,0.5,0,0,1.5,0,0,0,0,1

set_backend backend=cpp
stroke_path p1=-0.2,0,0.25 p2=0.2,0,0.25 steps=8
dump_state out=draw_viewrepeat_tex_cpp.json

undo

set_backend backend=wgsl
stroke_path p1=-0.2,0,0.25 p2=0.2,0,0.25 steps=8
dump_state out=draw_viewrepeat_tex_wgsl.json
