# A/B script: texdraw stroke under the cpp then wgsl backend gate.
# texdraw is a draw variant modulated by an inline `texture Rings { eval }`
# block (source/brush/kernels/texdraw.sbrush). The eval is a pure function
# of position/normal using the sin/cos/floor/fract intrinsics, so both
# backends must produce the same texRingsEval / tex_rings_eval result and
# the dumps must match exactly. The cpp dump is also diffed against
# tests/golden/texdraw.json by `make.mjs sbrush-verify`.
make_cube subdivs=16 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=texdraw
set_brush radius=0.25 strength=0.5

set_backend backend=cpp
stroke origin=0,0,0.25 normal=0,0,1
dump_state out=texdraw_cpp.json

undo

set_backend backend=wgsl
stroke origin=0,0,0.25 normal=0,0,1
dump_state out=texdraw_wgsl.json
