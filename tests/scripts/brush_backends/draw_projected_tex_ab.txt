# A/B script: PROJECTED texture coord space, cpp vs wgsl backend gate.
# Unlike the matrix-driven modes (Global/ViewPlane/ViewRepeat) this is the one
# coord space that consumes the surface normal: uv = (co - surfacePos) resolved
# in a deterministic orthonormal basis built from the brush-center surfaceNo.
# We stroke the +X cube face (normal ~+X) so |n.z| < 0.999 and the basis takes
# the (0,0,1) reference-axis branch — the common, non-degenerate path. cpp and
# wgsl key off the same ctx surfaceNo/surfacePos, so parity here proves the
# tangent-basis construction matches across backends. See draw_ab.txt for the
# A/B harness pattern.
make_cube subdivs=16 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=draw
set_brush radius=0.15 strength=6.0
set_texture pattern=rampx width=64 height=64
set_coord_space space=projected

set_backend backend=cpp
stroke_path p1=0.25,-0.2,0 p2=0.25,0.2,0 steps=8
dump_state out=draw_projected_tex_cpp.json

undo

set_backend backend=wgsl
stroke_path p1=0.25,-0.2,0 p2=0.25,0.2,0 steps=8
dump_state out=draw_projected_tex_wgsl.json
