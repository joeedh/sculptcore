# A/B script: GLOBAL texture coord space, cpp vs wgsl backend gate.
# Complements stroke_curved_ab.txt (which covers STROKE_CURVED): here the rampx
# texture is sampled in GLOBAL space (uv = co.xy), so the GPU exercises the real
# brush-texture upload + in-shader bilinear against the C++ sampleTexBilinear.
# See draw_ab.txt for the A/B harness pattern.
make_cube subdivs=16 size=0.5
build_spatial leaf_limit=256 depth_limit=8
set_brush_tool tool=draw
set_brush radius=0.15 strength=6.0
set_texture pattern=rampx width=64 height=64
set_coord_space space=global

set_backend backend=cpp
stroke_path p1=-0.2,0,0.25 p2=0.2,0,0.25 steps=8
dump_state out=draw_global_tex_cpp.json

undo

set_backend backend=wgsl
stroke_path p1=-0.2,0,0.25 p2=0.2,0,0.25 steps=8
dump_state out=draw_global_tex_wgsl.json
